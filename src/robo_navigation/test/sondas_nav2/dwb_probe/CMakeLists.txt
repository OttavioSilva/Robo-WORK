cmake_minimum_required(VERSION 3.8)
project(dwb_probe)
find_package(ament_cmake REQUIRED)
find_package(dwb_core REQUIRED)
find_package(dwb_plugins REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(dwb_probe probe.cpp)
ament_target_dependencies(dwb_probe dwb_core dwb_plugins nav2_core nav2_util pluginlib rclcpp)
find_package(nav2_behavior_tree REQUIRED)
find_package(behaviortree_cpp REQUIRED)
add_executable(bt_probe bt_probe.cpp)
ament_target_dependencies(bt_probe nav2_behavior_tree behaviortree_cpp)
