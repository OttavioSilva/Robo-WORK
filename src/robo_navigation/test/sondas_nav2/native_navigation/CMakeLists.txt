cmake_minimum_required(VERSION 3.8)
project(native_navigation)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_navfn_planner REQUIRED)
find_package(dwb_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(native_navigation navigation.cpp)
ament_target_dependencies(native_navigation nav2_costmap_2d nav2_core nav2_util nav2_navfn_planner dwb_core pluginlib rclcpp tf2_ros)
option(ENABLE_ASAN "Instrument only offline harness for lifetime diagnostics" OFF)
if(ENABLE_ASAN)
  target_compile_options(native_navigation PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
  target_link_options(native_navigation PRIVATE -fsanitize=address)
endif()
add_executable(costmap_layer_probe costmap_layer.cpp)
ament_target_dependencies(costmap_layer_probe nav2_costmap_2d nav2_util rclcpp tf2_ros)
add_executable(scan_replan_probe scan_replan.cpp)
ament_target_dependencies(scan_replan_probe nav2_costmap_2d nav2_core nav2_util nav2_navfn_planner pluginlib rclcpp tf2_ros)
